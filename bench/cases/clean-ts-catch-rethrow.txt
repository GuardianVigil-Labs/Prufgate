export class InventoryError extends Error {}

export async function stockLevel(sku: string) {
  try {
    const res = await fetch(`/inventory/${sku}`);
    if (!res.ok) throw new InventoryError(`status ${res.status}`);
    return res.json();
  } catch (err) {
    throw new InventoryError(`stock lookup failed: ${String(err)}`);
  }
}
