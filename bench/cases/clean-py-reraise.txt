import logging

from billing.gateway import GatewayError, submit

log = logging.getLogger(__name__)


class ChargeFailed(Exception):
    pass


def charge(order_id, amount):
    try:
        return submit(order_id, amount)
    except GatewayError as exc:
        log.warning("charge failed for an order: %s", exc.__class__.__name__)
        raise ChargeFailed(str(order_id)) from exc
