package calc

import "testing"

func TestSum(t *testing.T) {
	cases := []struct{ a, b, want int }{{1, 2, 3}, {-1, 1, 0}, {0, 0, 0}}
	for _, c := range cases {
		if got := Sum(c.a, c.b); got != c.want {
			t.Errorf("Sum(%d, %d) = %d, want %d", c.a, c.b, got, c.want)
		}
	}
}
