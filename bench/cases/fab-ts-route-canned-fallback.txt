import { NextResponse } from "next/server";

export async function GET() {
  try {
    const res = await fetch(process.env.RATES_URL as string);
    return NextResponse.json(await res.json());
  } catch {
    return NextResponse.json({ rates: [{ code: "EUR", rate: 1.08 }] });
  }
}
