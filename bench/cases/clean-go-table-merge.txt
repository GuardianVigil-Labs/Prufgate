func RoundCents(c int) int {

}
-- head/money/round_test.go --
package money

import "testing"

func TestRoundCents(t *testing.T) {
	cases := []struct {
		name string
		in   int
		want int
	}{
		{"down", 149, 100},
		{"up", 150, 200},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			if got := RoundCents(tc.in); got != tc.want {
				t.Fatalf("got %d, want %d", got, tc.want)
			}
		})
	}
}
