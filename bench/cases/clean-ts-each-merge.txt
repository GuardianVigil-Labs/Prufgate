export const slug = (s: string) => s.trim().toLowerCase().replace(/\s+/g, "-");
-- base/src/slug.test.ts --
import { slug } from "./slug";

test("lowercases", () => {
  expect(slug("Shoes")).toBe("shoes");
});

test("joins words", () => {
  expect(slug("red shoes")).toBe("red-shoes");
});

test("trims", () => {
  expect(slug(" shoes ")).toBe("shoes");
});
