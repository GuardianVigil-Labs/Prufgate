package api

func Version() string {
	return "1"
}

func Register(r *gin.Engine) {
	r.GET("/api/documents/:id", RequireAuth(), func(c *gin.Context) {
		var doc Document
		tenantID := c.GetString("tenant_id")
		db.Where("id = ? AND tenant_id = ?", c.Param("id"), tenantID).First(&doc)
		c.JSON(200, doc)
	})
}
