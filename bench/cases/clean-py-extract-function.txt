def invoice_total(lines, tax_rate):
    subtotal = 0
    for line in lines:
        subtotal += line["qty"] * line["price"]

def _subtotal(lines):
    return sum(line["qty"] * line["price"] for line in lines)


def invoice_total(lines, tax_rate):
    return round(_subtotal(lines) * (1 + tax_rate), 2)
